package Capstone_Project;



import org.openqa.selenium.By;

import org.openqa.selenium.WebDriver;

import org.openqa.selenium.WebElement;

import org.openqa.selenium.chrome.ChromeDriver;

import org.openqa.selenium.chrome.ChromeOptions;

import org.openqa.selenium.support.ui.ExpectedConditions;

import org.openqa.selenium.support.ui.WebDriverWait;

import org.testng.Assert;

import org.testng.annotations.AfterMethod;

import org.testng.annotations.BeforeMethod;

import org.testng.annotations.DataProvider;

import org.testng.annotations.Test;



import java.time.Duration;

import java.util.HashMap;

import java.util.Map;



public class Capstone {

WebDriver driver;

WebDriverWait wait;



//-----------Locators------

By usernameField = By.id("user-name");

By passwordField = By.id("password");

By loginButton = By.id("login-button");

By errorMessage = By.xpath("//h3[@data-test='error']");



@BeforeMethod

public void setUp() {

// Fix 1: Disable Chrome Password Manager & Autofill popups

ChromeOptions options = new ChromeOptions();

Map<String, Object> prefs = new HashMap<>();

prefs.put("credentials_enable_service", false);

prefs.put("profile.password_manager_enabled", false);

options.setExperimentalOption("prefs", prefs);

options.setExperimentalOption("excludeSwitches", new String[]{"enable-automation"});



driver = new ChromeDriver(options);

wait = new WebDriverWait(driver, Duration.ofSeconds(10));

driver.manage().window().maximize();

}



@DataProvider(name = "validData")

public Object[][] getValidData() {

return new Object[][]{

{"standard_user", "secret_sauce"}

};

}



@DataProvider(name = "invalidData")

public Object[][] getInvalidData() {

return new Object[][]{

{"user", "secret_sauce"}

};

}



@Test(priority = 1, dataProvider = "validData")

public void testPositiveLoginSauceDemo(String username, String password) {

driver.get("https://www.saucedemo.com/");



WebElement usernameFieldElement = wait.until(ExpectedConditions.visibilityOfElementLocated(usernameField));

driver.findElement(passwordField).sendKeys(password);

usernameFieldElement.sendKeys(username);

driver.findElement(loginButton).click();



wait.until(ExpectedConditions.urlContains("inventory.html"));

Assert.assertTrue(driver.getCurrentUrl().contains("inventory.html"), "Login Failed. URL: " + driver.getCurrentUrl());

}



@Test(priority = 2, dataProvider = "invalidData")

public void testNegativeLoginSauceDemo(String username, String password) {

driver.get("https://www.saucedemo.com/");



WebElement usernameFieldElement = wait.until(ExpectedConditions.visibilityOfElementLocated(usernameField));

driver.findElement(passwordField).sendKeys(password);

usernameFieldElement.sendKeys(username);

driver.findElement(loginButton).click();



WebElement errorBox = wait.until(ExpectedConditions.visibilityOfElementLocated(errorMessage));

Assert.assertEquals(errorBox.getText(), "Epic sadface: Username and password do not match any user in this service");

}



@Test(priority = 3, dataProvider = "validData")

public void Add3ProductsAndCheck(String username, String password) throws InterruptedException {

driver.get("https://www.saucedemo.com/");



WebElement usernameFieldElement = wait.until(ExpectedConditions.visibilityOfElementLocated(usernameField));

driver.findElement(passwordField).sendKeys(password);

usernameFieldElement.sendKeys(username);

driver.findElement(loginButton).click();



wait.until(ExpectedConditions.urlContains("inventory.html"));



//-----------------Add 3 Products in the carts

driver.findElement(By.id("add-to-cart-sauce-labs-backpack")).click();

driver.findElement(By.id("add-to-cart-sauce-labs-bike-light")).click();

driver.findElement(By.id("add-to-cart-sauce-labs-bolt-t-shirt")).click();



//-----------REMOVE 1 PRODUCT FROM CART and CHECK KART CONTENTS SHOULD BE 2

WebElement Badge = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("shopping_cart_badge")));

String Count = Badge.getText();

Assert.assertEquals(Count, "3", "Cart count is incorrect Expected is 3");



if (Count.equals("3")) {

System.out.println("2. 3 items are added ");

} else {

System.out.println("2. Items are not as per the expected 3 count");

}



driver.findElement(By.id("remove-sauce-labs-bolt-t-shirt")).click();

wait.until(ExpectedConditions.textToBePresentInElementLocated(By.className("shopping_cart_badge"), "2"));



String DecCount = driver.findElement(By.className("shopping_cart_badge")).getText();

Assert.assertEquals(DecCount, "2", "Cart count is incorrect Expected is 2");



if (DecCount.equals("2")) {

System.out.println("3. 2 items are in the cart after 1 removal ");

} else {

System.out.println("3. Items are not as per the expected 2 count");

}



//-------------CHECKOUT ITEMS FROM CART---------------

driver.findElement(By.className("shopping_cart_link")).click();



WebElement carttxt = wait.until(ExpectedConditions.visibilityOfElementLocated(By.xpath("//*[@id='header_container']/div[2]/span")));



if (carttxt.getText().equals("Your Cart")) {

System.out.println("3. Cart page is loaded");

} else {

System.out.println("3. Cart Page Not loaded ");

}



driver.findElement(By.id("checkout")).click();



WebElement checkinfo = wait.until(ExpectedConditions.visibilityOfElementLocated(By.xpath("//*[@id='header_container']/div[2]/span")));

if (checkinfo.getText().equals("Checkout: Your Information")) {

System.out.println("4. Checkout Information page is loaded");

} else {

System.out.println("4. Checkout Information Page Not loaded ");

}



driver.findElement(By.id("first-name")).sendKeys("Zadyaaaaa");

driver.findElement(By.id("last-name")).sendKeys("Bhauuuuu");

driver.findElement(By.id("postal-code")).sendKeys("411014");



driver.findElement(By.id("continue")).click();



WebElement checkoverview = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("title")));





if (checkoverview.getText().equals("Checkout: Overview")) {

System.out.println("4. Checkout Overview page is loaded with all the payment, shipping and price details");

} else {

System.out.println("4. Checkout Overview page is Not loaded ");

}

Thread.sleep(3000);



WebElement finishBtn = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("finish")));

finishBtn.click();



WebElement Finaldiag = wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("complete-header")));

if (Finaldiag.getText().equals("Thank you for your order!")) {

System.out.println("4. Order is placed and Thank you message is posted ");

} else {

System.out.println("4. Order is not placed as there is no thank you message ");

}

}



@AfterMethod

public void tearDown() {

if (driver != null) {

driver.quit();

}

}

}
